-- Player Profiles (Minimal data, mutable username, avatar & Bristol BS outcode)
create table if not exists profiles (
  id uuid references auth.users on delete cascade primary key,
  is_resident boolean default false,
  username text unique,
  email text unique,
  avatar text default '🧭',
  avatar_type text default 'emoji',
  outcode text check (outcode ~ '^BS[0-9]{1,2}$'),
  age_bracket text check (age_bracket in ('under_18', '18_24', '25_34', '35_44', '45_54', '55_plus')),
  total_discovered int default 0,
  weekly_discovered int default 0,
  monthly_discovered int default 0,
  is_flagged boolean default false,
  updated_at timestamptz default now()
);

-- Player Inventory (Tracks Tier 1 & Tier 2 items + unlocked recipes)
create table if not exists inventory (
  user_id uuid references profiles(id) on delete cascade,
  item_key text not null,
  quantity int default 0 check (quantity >= 0),
  discovered_recipe boolean default false,
  primary key (user_id, item_key)
);

-- Preloaded Daily Content (Synced to UK midnight)
create table if not exists daily_schedule (
  active_date date primary key,
  puzzle_question text not null,
  puzzle_options jsonb not null,
  puzzle_answer_index int not null,
  poll_question text,
  poll_options jsonb,
  challenge_text text not null
);

-- Ephemeral Poll Votes (Zero cross-day user tracking)
create table if not exists poll_votes (
  active_date date references daily_schedule(active_date) on delete cascade,
  option_index int not null,
  vote_count int default 0,
  primary key (active_date, option_index)
);

-- Consolidated poll votes table for plotting
create table if not exists poll_results_archive (
  poll_date date primary key,
  poll_question text not null,
  results jsonb not null, -- Format: [{"option": "...", "votes": 12, "percentage": 60.0}, ...]
  total_votes int not null default 0,
  archived_at timestamptz default now()
);

-- Username Moderation Reports
create table if not exists username_reports (
  id bigint generated by default as identity primary key,
  reported_user_id uuid references profiles(id) on delete cascade,
  reporter_id uuid references profiles(id),
  created_at timestamptz default now()
);

-- Anti-Cheat Telemetry (Server-only; hidden from client reads)
create table if not exists player_telemetry (
  user_id uuid references profiles(id) on delete cascade primary key,
  last_lat double precision,
  last_lng double precision,
  last_claim_at timestamptz
);

-- Server-Side Forge Recipes
create table if not exists forge_recipes (
  result_item_key text primary key,
  ingredient_1_key text not null,
  ingredient_1_qty int not null check (ingredient_1_qty > 0),
  ingredient_2_key text not null,
  ingredient_2_qty int not null check (ingredient_2_qty > 0)
);

-- Blocked Words for Username Validation
create table if not exists blocked_words (
  word text primary key
);

-- Table for upcoming timed collectibles
create table if not exists scheduled_collectibles (
  id bigint generated by default as identity primary key,
  title text not null,
  item_key text not null, -- e.g. 'stone', 'wood', 'water', 'metal', or special event key
  area_name text,         -- e.g. 'Brandon Hill (BS1)' or 'St George Park (BS5)'
  starts_at timestamptz not null,
  ends_at timestamptz not null,
  description text
);

-- Bulletin schedule
create table if not exists bulletin_schedule (
  active_date date primary key,
  title text not null,
  bulletin_html text not null
);

-- Create table for bug reports & improvement suggestions
create table if not exists game_feedback (
  id bigint generated by default as identity primary key,
  user_id uuid references auth.users on delete set null,
  category text not null default 'bug' check (category in ('bug', 'improvement', 'other')),
  message text not null check (char_length(trim(message)) between 5 and 1000),
  app_context jsonb, -- Stores optional device/browser/version context
  status text not null default 'open' check (status in ('open', 'in_progress', 'resolved')),
  created_at timestamptz default now()
);

-- 3. LEADERBOARD VIEWS
-- ============================================================================

-- Individual Resident Leaderboard
create or replace view leaderboard_individual as
select
  username,
  avatar,
  avatar_type,
  outcode,
  weekly_discovered,
  monthly_discovered,
  total_discovered
from profiles
where is_resident = true
  and username is not null
  and is_flagged = false
order by total_discovered desc
limit 100;

-- Bristol Outcode Leaderboard (Aggregated by BS postcode)
create or replace view leaderboard_outcodes as
select
  outcode,
  count(id) as active_residents,
  sum(weekly_discovered) as weekly_points,
  sum(monthly_discovered) as monthly_points,
  sum(total_discovered) as total_points
from profiles
where is_resident = true
  and outcode is not null
  and is_flagged = false
group by outcode
order by total_points desc;

-- A. Safe Profile & Resident Upgrade
drop function if exists update_player_profile(text, text, text, text, text, text, boolean);

create or replace function update_player_profile(
  p_username text,
  p_email text,
  p_avatar text,
  p_avatar_type text,
  p_outcode text,
  p_age_bracket text,
  p_is_resident boolean
)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  clean_username text := trim(p_username);
  clean_outcode text := upper(trim(coalesce(p_outcode, '')));
  clean_email text := lower(trim(nullif(p_email, '')));
begin
  if clean_outcode = '' or clean_outcode = 'VISITOR' then
    clean_outcode := null;
  end if;

  if clean_username !~ '^[a-zA-Z0-9_]{3,20}$' then
    raise exception 'Username must be 3-20 characters (letters, numbers, underscores).';
  end if;

  if exists (
    select 1 from blocked_words
    where clean_username ilike '%' || word || '%'
  ) then
    raise exception 'Username contains restricted words.';
  end if;

  if clean_outcode is not null and clean_outcode !~ '^BS[0-9]{1,2}$' then
    raise exception 'Invalid Bristol postcode outcode.';
  end if;

  -- Explicit duplicate email check against profiles
  if clean_email is not null and exists (
    select 1 from profiles
    where lower(email) = clean_email
      and id <> auth.uid()
  ) then
    raise exception 'This email address is already linked to another Resident account.';
  end if;

  insert into profiles (id, username, email, avatar, avatar_type, outcode, age_bracket, is_resident, updated_at)
  values (
    auth.uid(),
    clean_username,
    clean_email,
    coalesce(p_avatar, '🧭'),
    coalesce(p_avatar_type, 'emoji'),
    clean_outcode,
    p_age_bracket,
    p_is_resident,
    now()
  )
  on conflict (id) do update set
    username = excluded.username,
    email = coalesce(excluded.email, profiles.email),
    avatar = excluded.avatar,
    avatar_type = excluded.avatar_type,
    outcode = excluded.outcode,
    age_bracket = excluded.age_bracket,
    is_resident = excluded.is_resident,
    updated_at = now();
end;
$$;

-- B. Anti-Cheat Collectible Claim (Supports Offline Timestamps & Returns Authoritative Inventory)
create or replace function claim_collectible(
  p_item_key text,
  p_lat double precision,
  p_lng double precision,
  p_claimed_at timestamptz default now()
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user_id uuid := auth.uid();
  v_last_lat double precision;
  v_last_lng double precision;
  v_last_time timestamptz;
  v_effective_time timestamptz;
  v_dist_meters double precision;
  v_seconds_elapsed double precision;
  v_speed_kmh double precision;
  v_inventory jsonb;
begin
  if v_user_id is null then
    raise exception 'Not authenticated.';
  end if;

  -- Clamp claimed_at so it cannot be in the future or older than 24 hours
  v_effective_time := least(coalesce(p_claimed_at, now()), now());
  if v_effective_time < now() - interval '24 hours' then
    v_effective_time := now();
  end if;

  insert into profiles (id) values (v_user_id) on conflict do nothing;

  select last_lat, last_lng, last_claim_at
  into v_last_lat, v_last_lng, v_last_time
  from player_telemetry
  where user_id = v_user_id;

  if v_last_time is not null then
    -- Ensure offline claims are replayed in chronological order
    if v_effective_time <= v_last_time then
      raise exception 'Claim timestamp out of order.';
    end if;

    v_seconds_elapsed := extract(epoch from (v_effective_time - v_last_time));
    
    if v_seconds_elapsed < 3 then
      raise exception 'Claiming too fast.';
    end if;

    v_dist_meters := st_distance(
      st_setsrid(st_makepoint(v_last_lng, v_last_lat), 4326)::geography,
      st_setsrid(st_makepoint(p_lng, p_lat), 4326)::geography
    );

    v_speed_kmh := (v_dist_meters / 1000.0) / (v_seconds_elapsed / 3600.0);

    if v_speed_kmh > 120 then
      raise exception 'Movement speed validation failed.';
    end if;
  end if;

  insert into player_telemetry (user_id, last_lat, last_lng, last_claim_at)
  values (v_user_id, p_lat, p_lng, v_effective_time)
  on conflict (user_id) do update set
    last_lat = excluded.last_lat,
    last_lng = excluded.last_lng,
    last_claim_at = excluded.last_claim_at;

  insert into inventory (user_id, item_key, quantity)
  values (v_user_id, p_item_key, 1)
  on conflict (user_id, item_key) do update set
    quantity = inventory.quantity + 1;

  update profiles set
    total_discovered = total_discovered + 1,
    weekly_discovered = weekly_discovered + 1,
    monthly_discovered = monthly_discovered + 1,
    updated_at = now()
  where id = v_user_id;

  select coalesce(jsonb_agg(jsonb_build_object(
    'item_key', item_key,
    'quantity', quantity,
    'discovered_recipe', discovered_recipe
  )), '[]'::jsonb)
  into v_inventory
  from inventory
  where user_id = v_user_id;

  return jsonb_build_object(
    'status', 'claimed',
    'item', p_item_key,
    'inventory', v_inventory
  );
end;
$$;

-- C. Server-Validated Forge Crafting (Returns Authoritative Inventory)
create or replace function craft_forge_item(p_result_item_key text)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user_id uuid := auth.uid();
  v_recipe record;
  v_qty_1 int;
  v_qty_2 int;
  v_inventory jsonb;
begin
  if v_user_id is null then
    raise exception 'Not authenticated.';
  end if;

  select * into v_recipe from forge_recipes where result_item_key = p_result_item_key;
  if not found then
    raise exception 'Invalid recipe.';
  end if;

  select quantity into v_qty_1 from inventory
  where user_id = v_user_id and item_key = v_recipe.ingredient_1_key
  for update;

  select quantity into v_qty_2 from inventory
  where user_id = v_user_id and item_key = v_recipe.ingredient_2_key
  for update;

  if coalesce(v_qty_1, 0) < v_recipe.ingredient_1_qty
     or coalesce(v_qty_2, 0) < v_recipe.ingredient_2_qty then
    raise exception 'Insufficient ingredients.';
  end if;

  update inventory set quantity = quantity - v_recipe.ingredient_1_qty
  where user_id = v_user_id and item_key = v_recipe.ingredient_1_key;

  update inventory set quantity = quantity - v_recipe.ingredient_2_qty
  where user_id = v_user_id and item_key = v_recipe.ingredient_2_key;

  insert into inventory (user_id, item_key, quantity, discovered_recipe)
  values (v_user_id, p_result_item_key, 1, true)
  on conflict (user_id, item_key) do update set
    quantity = inventory.quantity + 1,
    discovered_recipe = true;

  select coalesce(jsonb_agg(jsonb_build_object(
    'item_key', item_key,
    'quantity', quantity,
    'discovered_recipe', discovered_recipe
  )), '[]'::jsonb)
  into v_inventory
  from inventory
  where user_id = v_user_id;

  return jsonb_build_object(
    'status', 'crafted',
    'item', p_result_item_key,
    'inventory', v_inventory
  );
end;
$$;

-- D. UK Midnight Daily Schedule Fetcher (Strips Polls for Under-18s)
create or replace function get_today_schedule()
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uk_date date := (now() at time zone 'Europe/London')::date;
  v_age text;
  v_row daily_schedule;
begin
  select * into v_row from daily_schedule where active_date = v_uk_date;
  if not found then
    return null;
  end if;

  select age_bracket into v_age from profiles where id = auth.uid();

  if v_age is null or v_age = 'under_18' then
    v_row.poll_question := null;
    v_row.poll_options := null;
  end if;

  return to_jsonb(v_row);
end;
$$;

-- E. Zero-Tracking Poll Vote Submission (18+ Residents Only)
create or replace function submit_poll_vote(p_option_index int)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uk_date date := (now() at time zone 'Europe/London')::date;
  v_is_resident boolean;
  v_age text;
begin
  select is_resident, age_bracket into v_is_resident, v_age
  from profiles where id = auth.uid();

  if coalesce(v_is_resident, false) = false or v_age is null or v_age = 'under_18' then
    raise exception 'Only registered residents aged 18+ can vote in daily polls.';
  end if;

  insert into poll_votes (active_date, option_index, vote_count)
  values (v_uk_date, p_option_index, 1)
  on conflict (active_date, option_index) do update set
    vote_count = poll_votes.vote_count + 1;
end;
$$;

-- F. Permanently Delete Player Profile & Cascading Data
create or replace function delete_my_account()
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user_id uuid := auth.uid();
begin
  if v_user_id is null then
    raise exception 'Not authenticated.';
  end if;

  -- Nullify reporter_id on reports filed by this user (no cascade on this FK)
  update username_reports set reporter_id = null where reporter_id = v_user_id;

  -- Deleting from profiles cascades to inventory, player_telemetry, and reported_user_id
  delete from profiles where id = v_user_id;
end;
$$;

grant execute on function delete_my_account() to authenticated;

-- ============================================================================
-- 6. RETAIN LOCAL INVENTORY WHEN FIRST UPGRADING TO RESIDENT AND REFRESH API SCHEMA CACHE
-- ============================================================================
create or replace function sync_local_inventory(
  p_items jsonb,
  p_total_discovered int default 0
)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user_id uuid := auth.uid();
  v_item jsonb;
  v_inv_sum int := 0;
  v_final_total int := 0;
  v_qty int;
  v_discovered boolean;
begin
  if v_user_id is null then
    raise exception 'Not authenticated.';
  end if;

  insert into profiles (id) values (v_user_id) on conflict do nothing;

  if p_items is not null and jsonb_typeof(p_items) = 'array' then
    for v_item in select * from jsonb_array_elements(p_items)
    loop
      v_qty := coalesce((v_item->>'quantity')::int, 0);
      v_discovered := coalesce((v_item->>'discovered_recipe')::boolean, false);

      if (v_item->>'item_key') is not null and (v_qty > 0 or v_discovered) then
        insert into inventory (user_id, item_key, quantity, discovered_recipe)
        values (
          v_user_id,
          v_item->>'item_key',
          greatest(v_qty, 0),
          v_discovered
        )
        on conflict (user_id, item_key) do update set
          quantity = greatest(inventory.quantity, excluded.quantity),
          discovered_recipe = inventory.discovered_recipe or excluded.discovered_recipe;
      end if;
    end loop;
  end if;

  select coalesce(sum(quantity), 0) into v_inv_sum
  from inventory
  where user_id = v_user_id;

  v_final_total := greatest(coalesce(p_total_discovered, 0), v_inv_sum);

  if v_final_total > 0 then
    update profiles set
      total_discovered = greatest(total_discovered, v_final_total),
      weekly_discovered = greatest(weekly_discovered, v_final_total),
      monthly_discovered = greatest(monthly_discovered, v_final_total),
      updated_at = now()
    where id = v_user_id;
  end if;
end;
$$;
